>     def __str__(self) -> str:
>         rows = (
>             ("who", self.name),
>             ("role", self.role),
>             ("now", self.building),
>             ("core", self.core),
>             ("edge", self.edge),
>         )
>         w = max(len(k) for k, _ in rows)
>         rule = "─" * 42
>         body = "\n".join(f"  {k:<{w}}  {v}" for k, v in rows)
>         return f"╭{rule}╮\n{body}\n╰{rule}╯"
